# SPEC 9: reading documents. Accepted forms are read through facts_from and
# printed back; rejected forms are given as goals and must stop the run.

=== canonical text reads back as the same term
spec: 8, 9
--- program.js
facts_from('data.js')
query(d(X))
--- data.js
d('plain')
d('it\'s')
d([1, [2, 3], ...T])
d(struct('hello world', 1))
d(add(1, mul(2, 3)))
d(mul(add(1, 2), 3))
d(pow(2, -1))
d(neg(pow(2, 2)))
d(pow(-2, 2))
d(neg(1))
d(neg(neg('x')))
d(or('a', and('b', 'c')))
d(and(lt(1, 2), ge(3, 4)))
d(1e+22)
d(-0.5)
--- stdout

d('it\'s')
d([1, [2, 3], ...A])
